cmake_minimum_required(VERSION 3.16)
project(carl CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -Wno-unused-parameter -Wno-unused-function)
find_package(OpenSSL REQUIRED)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
find_package(Boost REQUIRED COMPONENTS system)
find_package(nlohmann_json REQUIRED)
include_directories(${CMAKE_SOURCE_DIR}/include)
add_executable(carl src/carl.cpp)
target_link_libraries(carl PRIVATE OpenSSL::SSL OpenSSL::Crypto CURL::libcurl Boost::system Threads::Threads nlohmann_json::nlohmann_json)
if(WIN32)
  target_link_libraries(carl PRIVATE winhttp bcrypt advapi32 ws2_32 wlanapi)
  add_definitions(-D_WIN32_WINNT=0x0601)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(MINIUPNPC REQUIRED miniupnpc)
  target_include_directories(carl PRIVATE ${MINIUPNPC_INCLUDE_DIRS})
  target_link_libraries(carl PRIVATE ${MINIUPNPC_LIBRARIES})
endif()
install(TARGETS carl RUNTIME DESTINATION bin)
